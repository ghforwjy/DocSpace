## 📋 场景 1: 用户认证完整流程

### 完整代码示例

```javascript
// 配置基础 URL
const baseUrl = 'http://localhost:8092';

/**
 * 检查是否已认证
 */
async function checkAuth() {
  const response = await fetch(`${baseUrl}/api/2.0/authentication`, {
    method: 'GET',
    credentials: 'include',
    headers: {
      'Accept': 'application/json'
    }
  });
  return await response.json();
}

/**
 * 用户登录
 */
async function login(userName, password) {
  const response = await fetch(`${baseUrl}/api/2.0/authentication`, {
    method: 'POST',
    credentials: 'include',
    headers: {
      'Content-Type': 'application/json',
      'Accept': 'application/json'
    },
    body: JSON.stringify({
      userName: userName,
      password: password,
      session: true
    })
  });

  const result = await response.json();
  
  if (response.ok) {
    if (result.sms) {
      console.log('需要短信验证');
      return { needSms: true, result: result };
    } else if (result.tfa) {
      console.log('需要两步验证');
      return { needTfa: true, result: result };
    } else {
      console.log('登录成功!');
      return { success: true, result: result };
    }
  } else {
    console.error('登录失败:', result);
    return { success: false, error: result };
  }
}

/**
 * 注销登录
 */
async function logout() {
  await fetch(`${baseUrl}/api/2.0/authentication/logout`, {
    method: 'POST',
    credentials: 'include'
  });
  console.log('已注销');
}

// 完整认证流程示例
async function authenticationExample() {
  console.log('--- 认证流程示例 ---');
  
  // 1. 检查当前认证状态
  const isAuth = await checkAuth();
  console.log('当前认证状态:', isAuth);
  
  if (!isAuth) {
    // 2. 登录
    const loginResult = await login('user@example.com', 'your-password');
    
    if (loginResult.success) {
      // 3. 再次检查认证状态
      const verifyAuth = await checkAuth();
      console.log('验证认证:', verifyAuth);
      
      // 登录成功后可以进行其他操作...
      
      // 4. 最后注销（可选）
      // await logout();
    }
  }
}

// 运行示例
authenticationExample();
```

    console.log('5. 获取下载链接...');
    const downloadUrl = await getDownloadUrl(doc1.id);
    console.log('下载链接:', downloadUrl);
    
    console.log('✅ 文件管理流程完成!');
    
  } catch (error) {
    console.error('❌ 文件管理流程失败:', error);
  }

## 📋 综合场景: 完整的独立前端应用流程

/**
 * 独立前端应用 - 完整示例
 * 
 * 包含所有主要功能：
 * - 认证登录
 * - 用户管理
 * - 文件管理
 * - 权限控制
 * - 房间协作
 * - 版本管理
 */
class DocSpaceApp {
  constructor(baseUrl) {
    this.baseUrl = baseUrl;
    this.currentUser = null;
  }

  /**
   * 初始化应用
   */
  async init() {
    console.log('初始化 DocSpace 应用...');
    
    // 检查认证状态
    const isAuth = await this.checkAuth();
    
    if (isAuth) {
      // 已认证，获取用户信息
      this.currentUser = await this.getCurrentUser();
      console.log('✅ 应用初始化完成，用户:', this.currentUser.displayName);
    } else {
      console.log('⚠️ 用户未登录');
    }
    
    return isAuth;
  }

  /**
   * 认证相关方法
   */
  async checkAuth() {
    const response = await fetch(`${this.baseUrl}/api/2.0/authentication`, {
      method: 'GET',
      credentials: 'include',
      headers: { 'Accept': 'application/json' }
    });
    return await response.json();
  }

  async login(userName, password) {
    const response = await fetch(`${this.baseUrl}/api/2.0/authentication`, {
      method: 'POST',
      credentials: 'include',
      headers: {
        'Content-Type': 'application/json',
        'Accept': 'application/json'
      },
      body: JSON.stringify({
        userName: userName,
        password: password,
        session: true
      })
    });
    
    const result = await response.json();
    
    if (response.ok && !result.sms && !result.tfa) {
      // 登录成功
      this.currentUser = await this.getCurrentUser();
      return { success: true };
    }
    
    return { success: false, result: result };
  }

  async logout() {
    await fetch(`${this.baseUrl}/api/2.0/authentication/logout`, {
      method: 'POST',
      credentials: 'include'
    });
    this.currentUser = null;
  }

   * 用户管理

      `${this.baseUrl}/api/2.0/people/@self`,

        headers: { 'Accept': 'application/json' }

    );
    return await response.json();
  }

        method: 'POST',
        credentials: 'include',

          'Content-Type': 'application/json',
          'Accept': 'application/json'

          file: { title: title }
        })
      }

        method: 'POST',

        headers: {

      if (!isAuth) {
        console.log('请先登录!');
        return;
      }

      // 2. 浏览文件

## 📝 常见问题与注意事项

### 1. CORS 问题
如果遇到跨域问题，确保：
- 前端运行在正确的域名
- 服务端 CORS 配置正确

### 2. Cookie 问题
确保所有请求都使用 `credentials: 'include'`

### 3. 错误处理
所有 API 调用都应该有 try-catch 错误处理

### 4. 权限检查
操作前确保用户有相应的权限

### 5. 基础 URL 配置
根据实际部署配置正确的 baseUrl
